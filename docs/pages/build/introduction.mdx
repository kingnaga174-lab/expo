import React, { useState } from 'react';
import { View, Text, TouchableOpacity, ScrollView, TextInput, StyleSheet, Alert, SafeAreaView } from 'react-native';

const REF = {
  male:     { n: 'Male',     hb: [13, 17],   pcv: [40, 50], wbc: [4000, 11000], mcv: [80, 100] },
  female:   { n: 'Female',   hb: [12, 15.5], pcv: [36, 46], wbc: [4000, 11000], mcv: [80, 100] },
  pregnant: { n: 'Pregnant', hb: [11, 14.5], pcv: [33, 44], wbc: [5000, 14000], mcv: [80, 100] },
  child:    { n: 'Child',    hb: [11, 15],   pcv: [34, 44], wbc: [5000, 15500], mcv: [75, 92] },
  infant:   { n: 'Infant',   hb: [10.5, 13.5], pcv: [33, 41], wbc: [6000, 17500], mcv: [70, 86] },
};

const EMPTY = { hb: '', pcv: '', rbc: '', mcv: '', wbc: '', neut: '', lymph: '', eos: '', plt: '' };

const num = (v) => {
  if (v === '' || v === null || v === undefined) return null;
  const n = parseFloat(String(v).replace(/,/g, '').trim());
  return isNaN(n) ? null : n;
};

const fmt = (n) => (n === null ? '--' : String(Math.round(n)).replace(/\B(?=(\d{3})+(?!\d))/g, ','));

function parseText(raw) {
  const t = raw.replace(/(\d),(?=\d)/g, '$1');
  const get = (re) => { const m = t.match(re); return m ? m[1] : ''; };
  let wbc = num(get(/(?:tlc|wbc|total\s*leuc?k?ocyte)[^0-9\n]{0,25}(\d+(?:\.\d+)?)/i));
  if (wbc !== null && wbc < 100) wbc = Math.round(wbc * 1000);
  let plt = '';
  const pm = t.match(/(?:platelet|plt)[^0-9\n]{0,25}(\d+(?:\.\d+)?)([^\n]{0,15})/i);
  if (pm) {
    let p = parseFloat(pm[1]);
    const u = (pm[2] || '').toLowerCase();
    if (!isNaN(p)) {
      if (u.indexOf('lakh') > -1 || u.indexOf('lac') > -1 || p < 10) p = Math.round(p * 100000);
      else if (p < 1000) p = Math.round(p * 1000);
      plt = String(p);
    }
  }
  return {
    hb: get(/(?:ha?emoglobin|\bhb\b)(?!\s*a1c)[^0-9\n]{0,20}(\d+(?:\.\d+)?)/i),
    pcv: get(/(?:pcv|hct|ha?ematocrit|packed\s*cell)[^0-9\n]{0,20}(\d+(?:\.\d+)?)/i),
    rbc: get(/(?:\brbc\b|red\s*blood)[^0-9\n]{0,20}(\d+(?:\.\d+)?)/i),
    mcv: get(/(?:\bmcv\b|mean\s*corpuscular\s*vol)[^0-9\n]{0,20}(\d+(?:\.\d+)?)/i),
    wbc: wbc !== null ? String(wbc) : '',
    neut: get(/(?:neutrophil|polymorph)[^0-9\n]{0,20}(\d+(?:\.\d+)?)/i),
    lymph: get(/(?:lymphocyte)[^0-9\n]{0,20}(\d+(?:\.\d+)?)/i),
    eos: get(/(?:eosinophil)[^0-9\n]{0,20}(\d+(?:\.\d+)?)/i),
    plt,
  };
}

function analyze(raw, ctx) {
  const r = REF[ctx.group];
  const hb = num(raw.hb), pcv = num(raw.pcv), rbc = num(raw.rbc), mcv = num(raw.mcv);
  const wbc = num(raw.wbc), neut = num(raw.neut), lymph = num(raw.lymph), eos = num(raw.eos), plt = num(raw.plt);
  const chk = (v, lo, hi, plo, phi) =>
    v === null ? '' : plo && v < plo ? 'PANIC LOW' : phi && v > phi ? 'PANIC HIGH' : v < lo ? 'LOW' : v > hi ? 'HIGH' : 'OK';

  const flags = {
    hb: chk(hb, r.hb[0], r.hb[1], 7, 20),
    pcv: chk(pcv, r.pcv[0], r.pcv[1], 21, 60),
    wbc: chk(wbc, r.wbc[0], r.wbc[1], 2000, 30000),
    mcv: chk(mcv, r.mcv[0], r.mcv[1]),
    plt: chk(plt, 150000, 450000, 50000, 1000000),
  };

  const anc = wbc !== null && neut !== null ? Math.round((wbc * neut) / 100) : null;
  const aec = wbc !== null && eos !== null ? Math.round((wbc * eos) / 100) : null;
  const nlr = neut !== null && lymph ? (neut / lymph).toFixed(2) : '--';
  const mentzer = mcv !== null && rbc ? (mcv / rbc).toFixed(1) : '--';

  const panics = [];
  const notes = [];
  if (plt !== null && plt < 50000) panics.push('CRITICAL LOW PLATELETS (' + fmt(plt) + '/uL): High bleed risk. Avoid NSAIDs & IM shots; monitor urgently.');
  if (hb !== null && hb < 7) panics.push('SEVERE ANEMIA (Hb ' + hb + '): Assess stability, active bleeding, or transfusion need.');
  if (anc !== null && anc < 500) panics.push('SEVERE NEUTROPENIA (ANC ' + fmt(anc) + '): High sepsis risk; urgent care.');
  if (aec !== null && aec > 1500) panics.push('HYPEREOSINOPHILIA (AEC ' + fmt(aec) + '): Screen DRESS syndrome, LFT, RFT.');

  if (plt !== null && plt < 150000) {
    if (ctx.fever || ctx.day3) {
      notes.push({
        lvl: plt < 50000 ? 'p' : 'w',
        t: pcv > r.pcv[1] ? 'Febrile Low Platelets + High PCV (Strong Dengue Sign)' : 'Febrile Thrombocytopenia',
        b: 'Suggests Dengue, Scrub Typhus, Malaria, or Viral Fever. Watch for warning signs (bleeding, vomiting, abdominal pain).',
        nx: 'Dengue NS1 (D1-5) / IgM (D5+) • Scrub IgM • Malaria • AST/ALT',
      });
    } else if (ctx.drug || ctx.rash) {
      notes.push({
        lvl: 'w',
        t: 'Suspected Drug-Induced Thrombocytopenia (DITP)',
        b: 'Stop suspected drug (e.g., ceftriaxone) immediately and verify count on peripheral smear.',
        nx: 'Manual Smear (rule out clumping) • Urine Routine • Creatinine',
      });
    } else {
      notes.push({
        lvl: 'w',
        t: 'Isolated Low Platelets — Verify First',
        b: 'Rule out lab EDTA platelet clumping before treating if patient has no fever or bleeding.',
        nx: 'Manual Peripheral Smear • Repeat in Citrate Tube',
      });
    }
  }

  if (ctx.rash && ctx.drug && (plt === null || plt >= 150000)) {
    notes.push({
      lvl: 'w',
      t: 'Drug Skin Reaction with Normal Platelets',
      b: 'Raised purple spots on legs/hands suggest cutaneous vasculitis; flat sharp dusky plaques suggest Fixed Drug Eruption.',
      nx: 'Urine Routine (RBCs/protein) • Creatinine • AST/ALT',
    });
  }

  if (hb !== null && hb < r.hb[0]) {
    if (mcv !== null && mcv < r.mcv[0]) {
      const thal = mentzer !== '--' && parseFloat(mentzer) < 13;
      notes.push({
        lvl: hb < 7 ? 'p' : 'w',
        t: thal ? 'Microcytic Anemia — Thalassemia Trait Likely (Mentzer ' + mentzer + ')' : 'Microcytic Anemia — Iron Deficiency Likely',
        b: thal ? 'Confirm with HPLC before giving empiric iron.' : 'Suggests iron deficiency or chronic disease.',
        nx: 'Serum Ferritin • Iron/TIBC • HPLC • Peripheral Smear',
      });
    } else if (mcv !== null && mcv > r.mcv[1]) {
      notes.push({ lvl: 'w', t: 'Macrocytic Anemia', b: 'Commonly B12/Folate deficiency, liver disease, or hypothyroidism.', nx: 'Vitamin B12 • Folate • LFT • TSH' });
    } else {
      notes.push({ lvl: hb < 7 ? 'p' : 'w', t: 'Normocytic Anemia', b: 'Check acute bleed, chronic inflammation, or renal cause.', nx: 'Retic Count • Smear • Creatinine' });
    }
  }

  if (wbc !== null && wbc > r.wbc[1] && anc !== null && anc > 7500) {
    notes.push({ lvl: 'w', t: 'Neutrophilic Leukocytosis', b: 'High WBC (' + fmt(wbc) + ') & ANC (' + fmt(anc) + ') favors bacterial infection or steroid use.', nx: 'CRP • Cultures • Smear' });
  }

  const hasVal = Object.keys(raw).some((k) => raw[k] !== '');
  if (hasVal && !panics.length && !notes.length) {
    notes.push({ lvl: 'i', t: 'No Acute CBC Red Flags', b: 'Entered parameters are within range for this patient group.', nx: '' });
  }
  return { flags, anc, aec, nlr, mentzer, panics, notes, hasVal };
}

export default function App() {
  const [cbc, setCbc] = useState(EMPTY);
  const [txt, setTxt] = useState('');
  const [showPaste, setShowPaste] = useState(false);
  const [ctx, setCtx] = useState({ group: 'female', fever: false, day3: false, rash: false, drug: false });
  const res = analyze(cbc, ctx);

  const runParse = () => {
    const p = parseText(txt);
    const n = Object.keys(p).filter((k) => p[k] !== '').length;
    if (!n) return Alert.alert('No values found', 'Include names like Hb, WBC, Platelets with numbers.');
    setCbc(p);
    setShowPaste(false);
  };

  const sample = () => {
    setCbc({ hb: '14.8', pcv: '47.5', rbc: '5.1', mcv: '88', wbc: '3100', neut: '42', lymph: '50', eos: '2', plt: '42000' });
    setCtx({ group: 'female', fever: true, day3: true, rash: true, drug: true });
  };

  return (
    <SafeAreaView style={s.bg}>
      <ScrollView contentContainerStyle={s.pad} keyboardShouldPersistTaps="handled">
        <Text style={s.h1}>CBC Clinical Guide</Text>
        <Text style={s.sub}>Indian OPD Analyzer • Offline Live Text Reader</Text>

        <View style={s.row}>
          <TouchableOpacity style={[s.btn, s.b1]} onPress={() => setShowPaste(!showPaste)}>
            <Text style={s.bt}>{showPaste ? 'Hide Paste' : 'Paste Report Text'}</Text>
          </TouchableOpacity>
          <TouchableOpacity style={[s.btn, s.b2]} onPress={sample}>
            <Text style={s.bt}>Sample Case</Text>
          </TouchableOpacity>
          <TouchableOpacity style={[s.btn, s.b3]} onPress={() => { setCbc(EMPTY); setTxt(''); }}>
            <Text style={s.bt}>Clear</Text>
          </TouchableOpacity>
        </View>

        {showPaste && (
          <View style={s.box}>
            <Text style={s.lbl}>Snap report in Camera → Copy via Live Text → Paste here:</Text>
            <TextInput style={s.area} multiline value={txt} onChangeText={setTxt} placeholder="Hb 12.4, WBC 3400, Platelets 1.1 Lakhs..." />
            <TouchableOpacity style={[s.btn, s.b1, { marginRight: 0 }]} onPress={runParse}>
              <Text style={s.bt}>Auto-Extract & Analyze</Text>
            </TouchableOpacity>
          </View>
        )}

        <Text style={s.sec}>1. Patient Group</Text>
        <ScrollView horizontal showsHorizontalScrollIndicator={false}>
          {Object.keys(REF).map((g) => (
            <TouchableOpacity key={g} style={[s.ch, ctx.group === g && s.chOn]} onPress={() => setCtx({ ...ctx, group: g })}>
              <Text style={[s.ct, ctx.group === g && s.ctOn]}>{REF[g].n}</Text>
            </TouchableOpacity>
          ))}
        </ScrollView>

        <Text style={s.sec}>2. Symptoms</Text>
        <View style={s.wrap}>
          {[['fever', 'Fever'], ['day3', 'Day 3+'], ['rash', 'Rash/Purple'], ['drug', 'Recent Drug']].map((x) => (
            <TouchableOpacity key={x[0]} style={[s.ch, ctx[x[0]] && s.chOn]} onPress={() => setCtx({ ...ctx, [x[0]]: !ctx[x[0]] })}>
              <Text style={[s.ct, ctx[x[0]] && s.ctOn]}>{x[1]}</Text>
            </TouchableOpacity>
          ))}
        </View>

        <Text style={s.sec}>3. CBC Values</Text>
        <View style={s.grid}>
          {[['hb','Hb (g/dL)'],['pcv','PCV (%)'],['wbc','WBC (/uL)'],['plt','Platelets (/uL)'],['mcv','MCV (fL)'],['rbc','RBC (M/uL)'],['neut','Neut (%)'],['lymph','Lymph (%)'],['eos','Eos (%)']].map((f) => {
            const fl = res.flags[f[0]];
            const bad = fl && fl !== 'OK';
            return (
              <View key={f[0]} style={[s.cell, bad && s.cellBad]}>
                <Text style={[s.lbl, bad && { color: '#B91C1C' }]}>{f[1]} {bad ? '[' + fl + ']' : ''}</Text>
                <TextInput style={s.inp} keyboardType="decimal-pad" placeholder="--" value={cbc[f[0]]} onChangeText={(v) => setCbc({ ...cbc, [f[0]]: v })} />
              </View>
            );
          })}
        </View>

        {res.hasVal && (
          <View style={{ marginTop: 8 }}>
            {res.panics.map((m, i) => <View key={i} style={s.panic}><Text style={s.bt}>ALERT: {m}</Text></View>)}
            <View style={s.der}>
              <Text style={s.dt}>ANC: {fmt(res.anc)}  •  AEC: {fmt(res.aec)}  •  NLR: {res.nlr}  •  Mentzer: {res.mentzer}</Text>
            </View>
            <Text style={s.sec}>4. Clinical Guidance</Text>
            {res.notes.map((n, i) => (
              <View key={i} style={[s.card, n.lvl === 'p' ? s.cp : n.lvl === 'w' ? s.cw : s.ci]}>
                <Text style={s.nt}>{n.t}</Text>
                <Text style={s.nb}>{n.b}</Text>
                {!!n.nx && <Text style={s.nx}>Next: {n.nx}</Text>}
              </View>
            ))}
          </View>
        )}
      </ScrollView>
    </SafeAreaView>
  );
}
